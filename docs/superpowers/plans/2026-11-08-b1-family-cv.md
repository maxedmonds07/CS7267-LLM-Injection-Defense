# B1-famcv Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Two new B1 TF-IDF runs that choose `C` by family-grouped cross-validation (and, in one, set the threshold from out-of-fold scores). They are reported beside the current B1 through the unchanged harness, and judged by a decision rule fixed before any run.

**Architecture:** `src/eval/family_cv.py` holds two pure functions: `family_folds` assigns every train_pool row a fold by attack family (BIPIA) or by group (PoisonedRAG), keeping twins together, and `cross_validate` runs a parameter grid over those folds and returns the choice, a per-fold table and out-of-fold scores. `src/eval/harness.py` gains `attack_family` in `COLUMNS`, seen/unseen family slices, and an optional preset threshold for `evaluate`. `src/models/train.py` gains `--selection {val,famcv,famcv-oof}`, which picks how TF-IDF is selected. Everything after fitting (evaluate, report CSV, MLflow) is shared with the current B1.

**Tech Stack:** Python 3.11, uv, make, pandas 3, numpy, scikit-learn 1.9, MLflow 3.16, pytest.

**Spec:** `docs/superpowers/specs/2026-11-08-b1-family-cv-design.md`. Read it before any task: the anti-goals and the decision rule bind every task.

# src/eval/family_cv.py
"""Family-grouped cross-validation for choosing a detector's settings on attack families it
was not fit on. Spec: docs/superpowers/specs/2026-11-08-b1-family-cv-design.md."""

import numpy as np
import pandas as pd


def family_folds(frame, k=5, seed=7267) -> np.ndarray:
    """One fold in [0, k) per row. A group never spans two folds. In a source with several
    attack families, each family goes whole into one fold: largest first into the fold with
    the fewest attack rows, ties to the lowest index. Other groups (one-family sources, groups
    with no attack) are shuffled with `seed` and dealt round-robin."""
    attacks = frame[frame.label == 1]
    per_group = attacks.groupby("group").attack_family.nunique()
    if (per_group > 1).any():
        raise ValueError(f"groups with more than one attack family: {list(per_group[per_group > 1].index[:5])}")

    group_fold = {}
    for source, rows in attacks.groupby("source"):
        sizes = rows.attack_family.value_counts()
        if len(sizes) < 2:
            continue
        load, family_fold = [0] * k, {}
        for family in sorted(sizes.index, key=lambda f: (-sizes[f], f)):
            target = min(range(k), key=lambda i: (load[i], i))
            family_fold[family] = target
            load[target] += int(sizes[family])
        group_fold.update({g: family_fold[f] for g, f in rows.groupby("group").attack_family.first().items()})

    rest = sorted(set(frame.group) - set(group_fold))
    order = np.random.default_rng(seed).permutation(len(rest))
    group_fold.update({rest[j]: n % k for n, j in enumerate(order)})

    folds = frame.group.map(group_fold).to_numpy(dtype=int)
    labels = frame.label.to_numpy()
    for f in range(k):
        if len(set(labels[folds == f])) < 2:
            raise ValueError(f"fold {f} needs both labels; use fewer folds or more groups")
    return folds
```

# src/eval/family_cv.py (append; add `from dataclasses import dataclass` and
# `from metrics import auroc` to the imports)
@dataclass
class CVResult:
    best: object
    mean_auroc: float
    table: pd.DataFrame
    oof_scores: np.ndarray


def cross_validate(make_detector, grid, frame, folds) -> CVResult:
    """For each value in grid and each fold: fit make_detector(value) on the other folds,
    score this fold. Pick the value with the highest mean held-out AUROC; ties go to the
    earlier grid value. oof_scores are the chosen value's held-out scores, one per row."""
    texts, labels = frame.text.to_numpy(), frame.label.to_numpy()
    records, oof = [], {}
    for i, value in enumerate(grid):
        oof[i] = np.full(len(frame), np.nan)
        for f in sorted(set(folds)):
            held = folds == f
            model = make_detector(value).fit(texts[~held], labels[~held])
            scores = model.predict_proba(texts[held])
            oof[i][held] = scores
            records.append({"param": value, "fold": int(f), "auroc": auroc(scores, labels[held]),
                            "n_benign": int((labels[held] == 0).sum()), "n_adversarial": int(labels[held].sum())})
    table = pd.DataFrame(records)
    means = [table.auroc[table.param == value].mean() for value in grid]
    best = int(np.nanargmax(means))
    return CVResult(grid[best], float(means[best]), table, oof[best])
```

Note: `table.param == value` with string params works because grid values are distinct. Use positions (`i`) for `oof`, so float grid values are never dict keys.

```python
# src/models/train.py
from family_cv import cross_validate, family_folds
from metrics import auroc, threshold_at_fpr

SELECTIONS = ("val", "famcv", "famcv-oof")
FAMCV_FOLDS = 5
FOLD_SEED = 7267


def fit_tfidf_famcv(data, *, oof, k=FAMCV_FOLDS, target_fpr=0.01):
    """Choose C by family-grouped CV (spec 2026-11-08). famcv: folds of train, fit on train,
    threshold left to evaluate (val). famcv-oof: folds of train + val, refit on both, and the
    threshold comes from the chosen C's out-of-fold benign scores."""
    frame = pd.concat([data.train, data.val], ignore_index=True) if oof else data.train.reset_index(drop=True)
    folds = family_folds(frame, k=k, seed=FOLD_SEED)
    cv = cross_validate(lambda c: TfidfDetector(C=c), TFIDF_C_GRID, frame, folds)
    detector = TfidfDetector(C=cv.best).fit(frame.text, frame.label)
    selection = "famcv-oof" if oof else "famcv"
    detector.name = f"tfidf-{selection}"
    detector.cv_artifacts = {
        "cv_table.csv": cv.table,
        "folds.csv": frame.assign(fold=folds)[["source", "group", "attack_family", "fold"]]
            .sort_values("attack_family").drop_duplicates("group"),
    }
    params = {"selection": selection, "k": k, "fold_seed": FOLD_SEED, "C": cv.best, "cv_auroc": cv.mean_auroc}
    if oof:
        params["threshold"] = threshold_at_fpr(cv.oof_scores[frame.label.to_numpy() == 0], target_fpr)
    return detector, params
```

- `fit_detector` gains `selection="val"`, `folds=FAMCV_FOLDS`, `target_fpr=0.01`. ```bash
PYTHONPATH=src/data:src/eval:src/models uv run python - <<'EOF'
import math
import pandas as pd
runs = {n: pd.read_csv(f"report/baselines/{f}.csv") for n, f in
        [("tfidf", "b1_tfidf_lr"), ("tfidf-famcv", "b1_tfidf_lr_famcv"), ("tfidf-famcv-oof", "b1_tfidf_lr_famcv_oof")]}
def get(run, s, col, eval_set="test"):
    r = runs[run]; return float(r[(r.eval_set == eval_set) & (r.slice == s)][col].iloc[0])
def auroc_ok(run):
    return (get(run, "bipia_family=unseen", "auroc") >= get("tfidf", "bipia_family=unseen", "auroc")
            and get(run, "all", "auroc") >= get("tfidf", "all", "auroc") - 0.01)
def wilson(p, n, z=1.96):
    c = (p + z*z/(2*n)) / (1 + z*z/n); h = z*math.sqrt(p*(1-p)/n + z*z/(4*n*n)) / (1 + z*z/n); return c-h, c+h
rule1 = auroc_ok("tfidf-famcv")
rule2 = (abs(get("tfidf-famcv-oof", "all", "fpr") - 0.01) < abs(get("tfidf-famcv", "all", "fpr") - 0.01)
         and auroc_ok("tfidf-famcv-oof"))
print("rule 1 (famcv for C):", rule1, "| rule 2 (oof threshold):", rule2)
for run, r in runs.items():
    for _, row in r[r.slice.isin(["all", "bipia_family=unseen", "bipia_family=seen"])].iterrows():
        lo, hi = wilson(row.fpr, row.n_benign) if row.n_benign else (float("nan"),) * 2
        print(f"{run:16} {row.eval_set:7} {row.slice:22} AUROC {row.auroc:.3f} [{row.auroc_lo:.3f}, {row.auroc_hi:.3f}]"
              f"  TPR {row.tpr:.3f}  FPR {row.fpr:.3f} [{lo:.3f}, {hi:.3f}]")
EOF
